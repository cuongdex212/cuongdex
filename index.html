async function sendToAIWithRetry(imageBase64, retriesLeft) { const resultDiv = document.getElementById('result'); const API_KEY = localStorage.getItem('cuongdex_apikey'); try { const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-3.8-flash:generateContent?key=${API_KEY}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [ { text: "Chỉ ghi đáp án: A, B, C hoặc D kèm 1 chữ." }, { inline_data: { mime_type: "image/jpeg", data: imageBase64.split(',')[1] } } ] }], generationConfig: { maxOutputTokens: 20, temperature: 0.0 } }) }); const data = await response.json(); // Tự động xử lý nếu dính lỗi Server bận (503) hoặc Vượt quá giới hạn (429 / Quota exceeded) if ((response.status === 503 || response.status === 429 || (data.error && (data.error.code === 503 || data.error.code === 429 || data.error.message.includes("quota") || data.error.message.includes("high demand")))) && retriesLeft > 0) { resultDiv.innerText = `⏳ Đang đợi API hết hạn ngạch, thử lại sau 4s (${retriesLeft})...`; await new Promise(resolve => setTimeout(resolve, 4000)); // Chờ 4 giây trước khi thử lại return sendToAIWithRetry(imageBase64, retriesLeft - 1); } if (data.candidates && data.candidates[0].content && data.candidates[0].content.parts[0].text) { const aiAnswer = data.candidates[0].content.parts[0].text; resultDiv.innerText = "👉 " + aiAnswer; } else if (data.error) { resultDiv.innerText = "Lỗi API: " + data.error.message; } else { resultDiv.innerText = "Không nhận được phản hồi. Thử lại!"; } } catch (err) { if (retriesLeft > 0) { resultDiv.innerText = `🔄 Đang kết nối lại (${retriesLeft})...`; await new Promise(resolve => setTimeout(resolve, 2000)); return sendToAIWithRetry(imageBase64, retriesLeft - 1); } resultDiv.innerText = "Lỗi kết nối: " + err.message; } }
